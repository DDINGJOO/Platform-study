# 22 kubectl run · expose (2026-11-21 12:03:01 KST)

## kubectl run myweb2 --image=nginx:1.25.3-alpine --port=80 --dry-run=client -o yaml
  apiVersion: v1
  kind: Pod
  metadata:
    labels:
      run: myweb2
    name: myweb2
    namespace: ch05
  spec:
    containers:
    - image: nginx:1.25.3-alpine
      name: myweb2
      ports:
      - containerPort: 80
      resources: {}
    dnsPolicy: ClusterFirst
    restartPolicy: Always
  status: {}

## kubectl expose pod myweb2 --name=myweb2-svc --port=8002 --target-port=80 --dry-run=client -o yaml (셀렉터를 어디서 가져오나)
  spec:
    ports:
    - port: 8002
      protocol: TCP
      targetPort: 80
    selector:
      run: myweb2
  status:
