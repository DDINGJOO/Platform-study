# 무한 로딩을 풀어 보려고 Headlamp 의 base URL 을 API 서버 프록시 경로로 맞춘 시도 (작업 세션의 명령 기록에서 옮김, kube-system 시절)
master$ helm upgrade headlamp headlamp/headlamp -n kube-system --reuse-values --set config.baseURL=/api/v1/namespaces/kube-system/services/http:headlamp:80/proxy
master$ kubectl -n kube-system get pod -l app.kubernetes.io/instance=headlamp -o jsonpath="{.items[0].spec.containers[0].args}"
["-in-cluster","-in-cluster-context-name=main","-plugins-dir=/headlamp/plugins","-session-ttl=86400","-base-url=/api/v1/namespaces/kube-system/services/http:headlamp:80/proxy"]
