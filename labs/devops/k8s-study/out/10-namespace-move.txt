# kube-system 에 두었던 Headlamp 와 admin-user 를 전용 네임스페이스 headlamp 로 옮긴다
master$ kubectl -n kube-system delete serviceaccount admin-user && kubectl delete clusterrolebinding admin-user
serviceaccount "admin-user" deleted from kube-system namespace
clusterrolebinding.rbac.authorization.k8s.io "admin-user" deleted

master$ helm uninstall headlamp -n kube-system
release "headlamp" uninstalled

master$ kubectl create namespace headlamp
namespace/headlamp created

master$ helm install headlamp headlamp/headlamp -n headlamp | head -6

NAMESPACE: headlamp
STATUS: deployed
REVISION: 1

master$ kubectl -n headlamp rollout status deploy/headlamp --timeout=180s

deployment "headlamp" successfully rolled out

master$ cat headlamp-admin-user.yaml ClusterRoleBinding-admin-user.yaml | grep -v '^#'
apiVersion: v1
kind: ServiceAccount
metadata:
  name: admin-user
  namespace: headlamp
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: admin-user
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: cluster-admin
subjects:
- kind: ServiceAccount
  name: admin-user
  namespace: headlamp

master$ kubectl apply -f headlamp-admin-user.yaml -f ClusterRoleBinding-admin-user.yaml

clusterrolebinding.rbac.authorization.k8s.io/admin-user created
