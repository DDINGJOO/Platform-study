   87  // Jitter returns a time.Duration between duration and duration + maxFactor *
   88  // duration.
   89  //
   90  // This allows clients to avoid converging on periodic behavior. If maxFactor
   91  // is 0.0, a suggested default value will be chosen.
   92  func Jitter(duration time.Duration, maxFactor float64) time.Duration {
   93  	if maxFactor <= 0.0 {
   94  		maxFactor = 1.0
   95  	}
   96  	wait := duration + time.Duration(jitterRand()*maxFactor*float64(duration))
   97  	return wait
