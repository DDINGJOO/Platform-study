# | #!/usr/bin/env bash
# | # 사용법: traffic.sh <초> <URL당 초당 요청 수> <URL>...
# | # 각 URL 에 초당 rps 번씩 curl 을 보낸다. 끝나면 보낸 요청 수를 찍는다.
# | dur=$1; rps=$2; shift 2
# | end=$(( $(date +%s) + dur )); rounds=0
# | interval=$(awk "BEGIN{print 1/$rps}")
# | while [ "$(date +%s)" -lt "$end" ]; do
# |   for u in "$@"; do curl -s -o /dev/null --max-time 2 "$u" & done
# |   rounds=$((rounds+1)); sleep "$interval"
# | done
# | wait
# | echo "$(date +%T) 끝: URL ${#}개 x ${rounds}번 = $(( rounds * $# ))건 (${dur}초, URL당 ${rps}/s)"
