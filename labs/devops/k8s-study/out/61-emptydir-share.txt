# 61 emptyDir 로 파일 나누기 (2026-11-15 21:48:11 KST)

## c1 이 두 곳에 파일을 쓴다
  $ kubectl exec share-files -c c1 -- sh -c "echo from-c1 > /www/a.txt; echo from-c1 > /shared/a.txt"

## c2 에서 본다
  c2 의 /www   : []
  c2 의 /shared: [a.txt ]  내용: from-c1

## 두 컨테이너의 /shared 마운트 (mount 표에서)
  [c1] /dev/vda1 /shared ext4
  [c2] /dev/vda1 /shared ext4
