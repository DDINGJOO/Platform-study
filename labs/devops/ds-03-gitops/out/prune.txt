# prune: git 에서 service.yaml 을 지웠을 때
prune=false: 0.3초 뒤 새 커밋 인식. app=('OutOfSync', 'Healthy'), Service 리소스 상태=[{'kind': 'Service', 'name': 'web', 'namespace': 'demo', 'requiresPruning': True, 'status': 'OutOfSync', 'version': 'v1'}]
  클러스터에 Service 남음: True
prune=true 로 바꾸고 124.6초 뒤 Service 삭제. app=('Synced', 'Healthy')
# 주: prune 토글은 Application 스펙만 바꾼다. 컨트롤러는 automated/selfHeal 이 켜질 때만 즉시 Git 비교를 요청하고(automatedSyncEnabled), prune 변경에는 요청하지 않는다. 그래서 다음 주기 새로고침(이번에는 약 2분 뒤)에 지워졌다.
