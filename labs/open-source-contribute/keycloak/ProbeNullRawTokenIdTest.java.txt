package org.keycloak.protocol.oidc.encode;

import java.util.HashMap;
import java.util.Map;
import java.util.stream.Collectors;

import org.keycloak.OAuth2Constants;
import org.keycloak.util.JsonSerialization;

import org.junit.Test;

public class ProbeNullRawTokenIdTest {

    @Test
    public void probe() throws Exception {
        DefaultTokenContextEncoderProviderFactory factory = new DefaultTokenContextEncoderProviderFactory();
        factory.init(null);
        factory.grantsByShortcuts = new HashMap<>();
        factory.grantsByShortcuts.put("ro", OAuth2Constants.PASSWORD);
        factory.grantsToShortcuts = factory.grantsByShortcuts.entrySet().stream()
                .collect(Collectors.toMap(Map.Entry::getValue, Map.Entry::getKey));
        DefaultTokenContextEncoderProvider provider = new DefaultTokenContextEncoderProvider(null, factory);

        try {
            AccessTokenContext ctx = new AccessTokenContext(AccessTokenContext.SessionType.ONLINE,
                    AccessTokenContext.TokenType.REGULAR, OAuth2Constants.PASSWORD, null);
            System.out.println("PROBE constructor: accepted, getRawTokenId()=" + ctx.getRawTokenId());
            System.out.println("PROBE encodeTokenId=" + provider.encodeTokenId(ctx));
        } catch (Exception e) {
            System.out.println("PROBE constructor: " + e);
        }
        try {
            AccessTokenContext ctx = JsonSerialization.readValue(
                    "{\"sessionType\":\"ONLINE\",\"tokenType\":\"REGULAR\",\"grantType\":\"password\",\"rawTokenId\":null}",
                    AccessTokenContext.class);
            System.out.println("PROBE json: accepted, encodeTokenId=" + provider.encodeTokenId(ctx));
        } catch (Exception e) {
            Throwable t = e;
            while (t.getCause() != null) t = t.getCause();
            System.out.println("PROBE json: " + e.getClass().getSimpleName() + " root=" + t);
        }
    }
}
